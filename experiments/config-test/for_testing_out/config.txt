[Library]

[Playback]
# When true: automatically selects the next local track after the queue
# empties. When false: playback stops when the queue is empty.
# Queue tracks always play regardless of this setting.

# Limit the number of local tracks auto-played after queue exhaustion.
autoplay_limit=false
# Max local tracks to auto-play when autoplay_limit=true (1-9999).

# How many seconds each seek keypress moves (1-300).
seek_seconds=5

# Show the real-time FFT visualizer panel.

# Rendering style. Currently only "bars" is supported.

# Height of the visualizer in terminal rows (1-20).
