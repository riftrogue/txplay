# Txplay Configuration
# File: ~/.config/txplay/config.txt

[Library]
# Directories scanned for audio files (MP3, WAV, FLAC).
# Repeat this key for multiple locations. Tilde expansion is supported.
music_path=~/Music
music_path=~/music
music_path=~/Songs
music_path=~/songs

[Playback]
# When true: automatically selects the next local track after the queue empties.
# When false: playback stops when the queue empties (queue always plays regardless).
autoplay=true
# Cap the number of local tracks auto-played in a session.
autoplay_limit=false
# Max local tracks to auto-play when autoplay_limit=true (1-9999).
autoplay_limit_value=10
# How many seconds each seek keypress moves (1-300).
seek_seconds=5

[Visualizer]
# Show the real-time FFT visualizer panel.
enabled=true
# Rendering style. Currently only "bars" is supported.
style=bars
# Height of the visualizer in terminal rows (1-20).
height=6

[Keybindings]
# Playback
play_pause=Space
next=n
previous=b
# Navigation
navigation_up=ArrowUp
navigation_down=ArrowDown
focus_next=Tab
focus_previous=ShiftTab
play=Enter
back=Escape
# General
search=/
refresh=r
quit=q
# Seeking
seek_forward=ArrowRight
seek_backward=ArrowLeft
# Queue
queue_add=a
queue_remove=d
queue_clear=c
